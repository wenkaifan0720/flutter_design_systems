Choose a delivery channel

#

Choose Email or Post under Delivery. The line beneath the group shows which channel will be used.

import 'package:cupertino_ui/cupertino_ui.dart';

class ChooseChannel extends StatefulWidget {
  const ChooseChannel({super.key});

  @override
  State<ChooseChannel> createState() => _ChooseChannelState();
}

class _ChooseChannelState extends State<ChooseChannel> {
  String channel = 'Email';

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 360),
    child: CupertinoListSection.insetGrouped(
      header: const Text('Delivery channel'),
      children: [
        for (final name in ['Email', 'Text message'])
          CupertinoListTile(
            title: Text(name),
            trailing: channel == name
                ? const Icon(
                    CupertinoIcons.check_mark,
                    semanticLabel: 'Selected',
                  )
                : null,
            onTap: () => setState(() => channel = name),
          ),
      ],
    ),
  );
}
