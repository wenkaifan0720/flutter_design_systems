Choose a layout

#

Switch between List and Grid. The caption names the active layout.

import 'package:flutter/widgets.dart';
import 'package:studio_ui/studio_ui.dart';

class ChooseLayout extends StatefulWidget {
  const ChooseLayout({super.key});

  @override
  State<ChooseLayout> createState() => _ChooseLayoutState();
}

class _ChooseLayoutState extends State<ChooseLayout> {
  Set<String> layout = {'List'};

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    children: [
      StudioToggleBank<String>(
        semanticLabel: 'Layout',
        options: const ['List', 'Grid'],
        labelOf: (layout) => layout,
        multiSelectionEnabled: false,
        emptySelectionAllowed: false,
        selected: layout,
        onSelectionChanged: (next) => setState(() => layout = next),
      ),
      const SizedBox(height: 12),
      Text('Layout: ${layout.single}'),
    ],
  );
}

Parameters

StudioToggleBank API ↗
semanticLabel
Names the group.
options
List and Grid, in reading order.
selected
Current layout.
onSelectionChanged
Receives the new selection.
multiSelectionEnabled
Allows a single layout when false.
emptySelectionAllowed
Keeps one layout selected when false.
