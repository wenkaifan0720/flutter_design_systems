Read sharing details

#

Open Sharing details to see who can read the file. Dismiss the note and open it once more with the keyboard.

Enter or Space opens it with the focus inside; Escape closes it and the focus goes back to Sharing details.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class ReadSharingDetails extends StatelessWidget {
  const ReadSharingDetails({super.key});

  @override
  Widget build(BuildContext context) => CivicPopover(
    title: 'Sharing details',
    content: const Text('Only invited people can read this file.'),
    builder: (context, anchor) => CivicButton.secondary(
      expand: false,
      onPressed: anchor.toggle,
      child: const Text('Sharing details'),
    ),
  );
}
