Load file details

#

Read the loading state, then choose Show details. The placeholder gives way to a file name and its owner. Choose Reload to try again.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class LoadFileDetails extends StatefulWidget {
  const LoadFileDetails({super.key});
  @override
  State<LoadFileDetails> createState() => _LoadFileDetailsState();
}

class _LoadFileDetailsState extends State<LoadFileDetails> {
  bool loading = true;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    crossAxisAlignment: CrossAxisAlignment.stretch,
    spacing: 16,
    children: [
      if (loading)
        const CivicSkeleton(
          label: 'Loading file details',
          child: CivicSkeletonText(lines: 2),
        )
      else
        const Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [Text('Planning notes'), Text('Owner: Mara Chen')],
        ),
      Align(
        child: CivicButton.secondary(
          expand: false,
          onPressed: () => setState(() => loading = !loading),
          child: Text(loading ? 'Show details' : 'Reload'),
        ),
      ),
    ],
  );
}
