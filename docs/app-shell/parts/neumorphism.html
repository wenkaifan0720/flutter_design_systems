Navigate a workspace

#

Choose Search, then switch between phone and desktop. Observe where the look places navigation and whether the current page stays selected.

import 'package:flutter/widgets.dart';
import 'package:neumorphism_ui/neumorphism_ui.dart';

class NavigateWorkspace extends StatefulWidget {
  const NavigateWorkspace({super.key});
  @override
  State<NavigateWorkspace> createState() => _NavigateWorkspaceState();
}

class _NavigateWorkspaceState extends State<NavigateWorkspace> {
  int selected = 0;
  static const destinations = [
    NeumorphicNavItem(
      label: 'Home',
      short: 'Home',
      icon: NeumorphicGlyphs.house,
    ),
    NeumorphicNavItem(
      label: 'Search',
      short: 'Search',
      icon: NeumorphicGlyphs.search,
    ),
  ];

  @override
  Widget build(BuildContext context) => NeumorphicScaffold(
    scroll: false,
    rail: [
      for (var i = 0; i < destinations.length; i++)
        NeumorphicRailLink(
          label: destinations[i].label,
          selected: i == selected,
          onPressed: () => setState(() => selected = i),
        ),
    ],
    footer: NeumorphicNavigationBar(
      items: destinations,
      selectedIndex: selected,
      onDestinationSelected: (next) => setState(() => selected = next),
    ),
    body: Center(child: Text(selected == 0 ? 'Home page' : 'Search page')),
  );
}

Parameters

NeumorphicScaffold API ↗
footer
Places the look’s destination bar alongside the page body.
body
Content of the selected destination.
scroll
Controls whether the scaffold scrolls the body.
rail
Navigation links for the wide rail.
