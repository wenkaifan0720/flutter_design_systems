Choose Search, then switch between phone and desktop. Observe where the look places navigation and whether the current page stays selected.

import 'package:flutter/widgets.dart';
import 'package:material_ui/material_ui.dart' show Icons;
import 'package:keycap_ui/keycap_ui.dart';

class NavigateWorkspace extends StatefulWidget {
  const NavigateWorkspace({super.key});
  @override
  State<NavigateWorkspace> createState() => _NavigateWorkspaceState();
}

class _NavigateWorkspaceState extends State<NavigateWorkspace> {
  int selected = 0;

  @override
  Widget build(BuildContext context) => KeycapScaffold.branches(
    header: const Text('Workspace'),
    destinations: const [
      KeycapDestination(icon: Icons.home_outlined, label: 'Home'),
      KeycapDestination(icon: Icons.search, label: 'Search'),
    ],
    selectedIndex: selected,
    onDestinationSelected: (next) => setState(() => selected = next),
    branches: [
      (context) => const KeycapScreen(body: Center(child: Text('Home page'))),
      (context) => const KeycapScreen(body: Center(child: Text('Search page'))),
    ],
  );
}

Parameters

KeycapScaffold API ↗
header
Names the app beside its navigation on wider windows.
destinations
The places, each an icon and a label, in reading order.
selectedIndex
The place on screen.
onDestinationSelected
Receives the place the reader chooses.
branches
Builds each place’s first page; each place keeps its own pages, and choosing it again returns to its first.
