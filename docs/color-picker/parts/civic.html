Choose an ink colour

#

Change the hue and then adjust the colour. The hexadecimal readout shows the red, green and blue channels of the new ink.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class ChooseInk extends StatefulWidget {
  const ChooseInk({super.key});
  @override
  State<ChooseInk> createState() => _ChooseInkState();
}

class _ChooseInkState extends State<ChooseInk> {
  final ink = DesignColorController(const Color(0xFF1D70B8));
  @override
  void dispose() {
    ink.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => SingleChildScrollView(
    child: Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.stretch,
      spacing: 16,
      children: [
        CivicColorPicker(labelText: 'Ink', controller: ink),
        ValueListenableBuilder(
          valueListenable: ink,
          builder: (context, _, _) {
            final rgb = ink.color.toARGB32() & 0xFFFFFF;
            return Text(
              'Ink: #${rgb.toRadixString(16).padLeft(6, '0').toUpperCase()}',
            );
          },
        ),
      ],
    ),
  );
}
