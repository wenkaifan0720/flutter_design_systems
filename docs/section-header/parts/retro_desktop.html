Browse the agenda

#

Read today’s agenda, then switch to the other day. The heading and the appointment beneath it change together.

components/section-header/retro_desktop/browse_days.dart
import 'package:flutter/widgets.dart';
import 'package:retro_desktop_ui/retro_desktop_ui.dart';

class BrowseDays extends StatefulWidget {
  const BrowseDays({super.key});
  @override
  State<BrowseDays> createState() => _BrowseDaysState();
}

class _BrowseDaysState extends State<BrowseDays> {
  bool tomorrow = false;
  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 440),
    child: SingleChildScrollView(
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          RetroDesktopSectionHeader(tomorrow ? 'Tomorrow' : 'Today'),
          const SizedBox(height: 12),
          Text(tomorrow ? '10 am · Walk' : '2 pm · Workshop'),
          const SizedBox(height: 12),
          RetroDesktopButton(
            tomorrow ? 'Show today' : 'Show tomorrow',
            onPressed: () => setState(() => tomorrow = !tomorrow),
          ),
        ],
      ),
    ),
  );
}
