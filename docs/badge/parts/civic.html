Read messages

#

Begin with three unread messages. Mark one read to see the count fall, then reset the inbox when it is empty.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class ReadMessages extends StatefulWidget {
  const ReadMessages({super.key});
  @override
  State<ReadMessages> createState() => _ReadMessagesState();
}

class _ReadMessagesState extends State<ReadMessages> {
  int unread = 3;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 16,
    children: [
      CivicBadge(count: unread, noun: 'message', child: const Text('Inbox')),
      Wrap(
        spacing: 12,
        runSpacing: 12,
        alignment: WrapAlignment.center,
        children: [
          CivicButton.secondary(
            expand: false,
            onPressed: unread == 0 ? null : () => setState(() => unread--),
            child: const Text('Mark one read'),
          ),
          CivicButton.secondary(
            expand: false,
            onPressed: unread == 0 ? () => setState(() => unread = 3) : null,
            child: const Text('Reset inbox'),
          ),
        ],
      ),
    ],
  );
}

Parameters

CivicBadge API ↗
count
How many there are; none shows no count.
noun
What one of them is, said with the count (“3 messages”).
nouns
The plural where adding an s is wrong.
child
What the count belongs to.
needsAction
Draws the count in the warning colour, for something to do.
max
The largest count drawn; more reads as “99+”.
