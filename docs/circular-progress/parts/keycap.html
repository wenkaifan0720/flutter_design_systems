Sync a folder

#

Start the folder sync. The example keeps it running until you choose Finish sync, so you can inspect the spinner at either text size.

components/circular-progress/keycap/sync_folder.dart
import 'package:flutter/widgets.dart';
import 'package:keycap_ui/keycap_ui.dart';

class SyncFolder extends StatefulWidget {
  const SyncFolder({super.key});
  @override
  State<SyncFolder> createState() => _SyncFolderState();
}

class _SyncFolderState extends State<SyncFolder> {
  bool running = false;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 12,
    children: [
      if (running) const KeycapSpinner(semanticLabel: 'Syncing folder'),
      Text(running ? 'Syncing folder' : 'Folder ready'),
      KeycapButton(
        onPressed: () => setState(() => running = !running),
        child: Text(running ? 'Finish sync' : 'Start sync'),
      ),
    ],
  );
}
