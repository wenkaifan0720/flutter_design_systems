Sort the stock list

#

Sort by Units to put the smallest quantity first, then reverse the order. Select a row or move to the next page to inspect the remaining record.

import 'package:material_ui/material_ui.dart';

class SortStock extends StatefulWidget {
  const SortStock({super.key});

  @override
  State<SortStock> createState() => _SortStockState();
}

class _SortStockState extends State<SortStock> {
  bool ascending = true;
  int column = 0;
  int page = 0;
  final selected = <String>{};
  static const stock = {'Ink': 12, 'Paper': 4, 'Pens': 8};

  void sort(int index, bool up) => setState(() {
    column = index;
    ascending = up;
    page = 0;
  });
  List<MapEntry<String, int>> get ordered => stock.entries.toList()
    ..sort(
      (a, b) =>
          (ascending ? 1 : -1) *
          (column == 0 ? a.key.compareTo(b.key) : a.value.compareTo(b.value)),
    );

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 540),
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 12,
      children: [
        SingleChildScrollView(
          scrollDirection: Axis.horizontal,
          child: DataTable(
            sortColumnIndex: column,
            sortAscending: ascending,
            columns: [
              DataColumn(label: const Text('Item'), onSort: sort),
              DataColumn(
                label: const Text('Units'),
                numeric: true,
                onSort: sort,
              ),
            ],
            rows: [
              for (final row in ordered.skip(page * 2).take(2))
                DataRow(
                  selected: selected.contains(row.key),
                  onSelectChanged: (value) => setState(() {
                    if (value == true) {
                      selected.add(row.key);
                    } else {
                      selected.remove(row.key);
                    }
                  }),
                  cells: [
                    DataCell(Text(row.key)),
                    DataCell(Text('${row.value}')),
                  ],
                ),
            ],
          ),
        ),
        TextButton(
          onPressed: () => setState(() => page = 1 - page),
          child: Text(page == 0 ? 'Next page' : 'Previous page'),
        ),
      ],
    ),
  );
}
