Name a project

#

Type a one-letter project name to see the validation message. Add a second letter to clear it.

components/form-field/minimal/name_project.dart
import 'package:flutter/widgets.dart';
import 'package:minimal_ui/minimal_ui.dart';

class NameProject extends StatefulWidget {
  const NameProject({super.key});
  @override
  State<NameProject> createState() => _NameProjectState();
}

class _NameProjectState extends State<NameProject> {
  bool tooShort = false;

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 400),
    child: SingleChildScrollView(
      child: MinimalFormField(
        label: 'Project name',
        help: 'For example, Field notes.',
        error: tooShort ? 'Use at least two letters.' : null,
        child: MinimalTextField(
          onChanged: (value) =>
              setState(() => tooShort = value.trim().length == 1),
        ),
      ),
    ),
  );
}
