Name a draft

#

Type a title, then edit or erase it. The draft heading below follows your changes and returns to Untitled when the field is empty.

components/text-field/material/name_draft.dart
import 'package:material_ui/material_ui.dart';

class NameDraft extends StatefulWidget {
  const NameDraft({super.key});

  @override
  State<NameDraft> createState() => _NameDraftState();
}

class _NameDraftState extends State<NameDraft> {
  String title = '';

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 360),
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 12,
      children: [
        TextField(
          decoration: const InputDecoration(labelText: 'Draft title'),
          onChanged: (value) => setState(() => title = value),
        ),
        Text(title.isEmpty ? 'Untitled' : title),
      ],
    ),
  );
}
