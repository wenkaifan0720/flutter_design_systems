Tag a draft

#

Add Home or Travel to the draft, then remove a label and read the updated list below.

Type to narrow the answers; the arrow keys move through them and Enter adds one. Each chosen answer has its own remove button.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class TagDraft extends StatefulWidget {
  const TagDraft({super.key});
  @override
  State<TagDraft> createState() => _TagDraftState();
}

class _TagDraftState extends State<TagDraft> {
  List<String> labels = ['Work'];

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    crossAxisAlignment: CrossAxisAlignment.stretch,
    spacing: 16,
    children: [
      CivicMultiSelect<String>(
        labelText: 'Labels',
        options: const ['Home', 'Travel', 'Work'],
        values: labels,
        onChanged: (next) => setState(() => labels = next),
      ),
      Text(labels.isEmpty ? 'No labels' : 'Draft labels: ${labels.join(', ')}'),
    ],
  );
}
