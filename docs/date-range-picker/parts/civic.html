Book a date range

#

Choose the first and last days of a stay. Try another pair to replace the original booking.

Each date is typed as day, month and year; Choose dates opens a calendar to pick both ends.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

const _months = [
  'January',
  'February',
  'March',
  'April',
  'May',
  'June',
  'July',
  'August',
  'September',
  'October',
  'November',
  'December',
];

String _date(DateTime day) =>
    '${day.day} ${_months[day.month - 1]} ${day.year}';

class BookDates extends StatefulWidget {
  const BookDates({super.key});
  @override
  State<BookDates> createState() => _BookDatesState();
}

class _BookDatesState extends State<BookDates> {
  DesignDateRange? stay;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    crossAxisAlignment: CrossAxisAlignment.stretch,
    spacing: 16,
    children: [
      CivicDateRangePicker(
        labelText: 'When is your stay?',
        value: stay,
        firstDate: DateTime(2026, 11),
        lastDate: DateTime(2026, 12, 31),
        onChanged: (range) => setState(() => stay = range),
      ),
      Text(switch (stay) {
        null => 'No dates chosen',
        DesignDateRange(:final start, end: null) => 'From ${_date(start)}',
        DesignDateRange(:final start, :final end?) =>
          'From ${_date(start)} to ${_date(end)}',
      }),
    ],
  );
}
