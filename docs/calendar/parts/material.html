Choose a day on the calendar

#

Choose a day in November, then browse to December and choose another. The date below the calendar follows the selection.

components/calendar/material/pick_day.dart
import 'package:material_ui/material_ui.dart';

class PickDay extends StatefulWidget {
  const PickDay({super.key});

  @override
  State<PickDay> createState() => _PickDayState();
}

class _PickDayState extends State<PickDay> {
  DateTime day = DateTime(2026, 11, 20);

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 420),
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 12,
      children: [
        CalendarDatePicker(
          initialDate: DateTime(2026, 11, 20),
          firstDate: DateTime(2026),
          lastDate: DateTime(2027, 12, 31),
          onDateChanged: (v) => setState(() => day = v),
        ),
        Text(MaterialLocalizations.of(context).formatFullDate(day)),
      ],
    ),
  );
}
