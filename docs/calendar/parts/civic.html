Choose a day on the calendar

#

Choose a day in November, then browse to December and choose another. The date below the calendar follows the selection.

Tab reaches the month’s days; the arrow keys move a day or a week, Page Up and Page Down a month, and Enter or Space chooses.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

const _months = [
  'January',
  'February',
  'March',
  'April',
  'May',
  'June',
  'July',
  'August',
  'September',
  'October',
  'November',
  'December',
];

String _date(DateTime day) =>
    '${day.day} ${_months[day.month - 1]} ${day.year}';

class PickDay extends StatefulWidget {
  const PickDay({super.key});
  @override
  State<PickDay> createState() => _PickDayState();
}

class _PickDayState extends State<PickDay> {
  DateTime chosen = DateTime(2026, 11, 12);
  late final dates = DesignCalendarController(
    firstDate: DateTime(2026, 11),
    lastDate: DateTime(2026, 12, 31),
    selectedDate: chosen,
  );
  @override
  void dispose() {
    dates.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 16,
    children: [
      CivicCalendar(
        controller: dates,
        onChanged: (day) => setState(() => chosen = day),
      ),
      Text('Chosen: ${_date(chosen)}'),
    ],
  );
}
