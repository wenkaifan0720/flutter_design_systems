Adjust zoom

#

Use Out or In to change the zoom by ten percentage points. The current value appears below the controls.

components/button-group/material/adjust_zoom.dart
import 'package:material_ui/material_ui.dart';

class AdjustZoom extends StatefulWidget {
  const AdjustZoom({super.key});

  @override
  State<AdjustZoom> createState() => _AdjustZoomState();
}

class _AdjustZoomState extends State<AdjustZoom> {
  int zoom = 100;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 12,
    children: [
      OverflowBar(
        spacing: 8,
        children: [
          OutlinedButton(
            onPressed: zoom > 50 ? () => setState(() => zoom -= 10) : null,
            child: const Text('Out'),
          ),
          OutlinedButton(
            onPressed: zoom < 200 ? () => setState(() => zoom += 10) : null,
            child: const Text('In'),
          ),
        ],
      ),
      Text('Zoom: $zoom%'),
    ],
  );
}
