Set a due date

#

Set a due date in November. Some looks open a calendar from a field, others display it in place, and Civic takes the date as three numbers.

Civic takes the day, month and year as three text fields. The date remains visible in those fields.

components/date-picker/civic/set_due_date.dart
import 'package:flutter/widgets.dart';
import 'package:civic_ui/civic_ui.dart';

class SetDueDate extends StatelessWidget {
  const SetDueDate({super.key});

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 400),
    child: SingleChildScrollView(
      child: const CivicDateInput(
        labelText: 'Due date',
        helperText: 'For example, 12 11 2026',
        dayLabel: 'Day',
        monthLabel: 'Month',
        yearLabel: 'Year',
      ),
    ),
  );
}
