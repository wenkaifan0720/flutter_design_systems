import 'package:cupertino_ui/cupertino_ui.dart';

class SetDueDate extends StatefulWidget {
  const SetDueDate({super.key});

  @override
  State<SetDueDate> createState() => _SetDueDateState();
}

class _SetDueDateState extends State<SetDueDate> {
  DateTime due = DateTime(2026, 11, 20);
  String dateLabel(DateTime date) =>
      '${date.year}-${date.month.toString().padLeft(2, '0')}-${date.day.toString().padLeft(2, '0')}';

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 450),
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 12,
      children: [
        SizedBox(
          height: 170,
          child: CupertinoDatePicker(
            mode: CupertinoDatePickerMode.date,
            initialDateTime: DateTime(2026, 11, 20),
            minimumYear: 2026,
            maximumYear: 2027,
            onDateTimeChanged: (v) => setState(() => due = v),
          ),
        ),
        Text('Due date: ${dateLabel(due)}'),
      ],
    ),
  );
}
