Add a day

#

Activate the plus to extend a loan by one day. The number below it shows the new loan period.

components/icon-button/minimal/add_a_day.dart
import 'package:flutter/widgets.dart';
import 'package:minimal_ui/minimal_ui.dart';
import 'package:flutter/cupertino.dart' show CupertinoIcons;

class AddADay extends StatefulWidget {
  const AddADay({super.key});

  @override
  State<AddADay> createState() => _AddADayState();
}

class _AddADayState extends State<AddADay> {
  int days = 7;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    children: [
      MinimalIconButton(
        CupertinoIcons.plus,
        label: 'Add a day',
        onPressed: () => setState(() => days++),
      ),
      const SizedBox(height: 12),
      Text('Loan period: $days days'),
    ],
  );
}

Parameters

MinimalIconButton API ↗
icon
Plus symbol for extending the loan.
label
Accessible name for the icon-only action.
onPressed
Adds one day to the current loan period.
