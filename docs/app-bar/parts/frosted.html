Save a note from the app bar

#

Use Save in the Notes bar. The status changes from Unsaved to Saved.

components/app-bar/frosted/save_note.dart
import 'package:flutter/widgets.dart';
import 'package:frosted_ui/frosted_ui.dart';

class SaveNote extends StatefulWidget {
  const SaveNote({super.key});
  @override
  State<SaveNote> createState() => _SaveNoteState();
}

class _SaveNoteState extends State<SaveNote> {
  bool saved = false;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    children: [
      FrostedAppBar(
        title: 'Notes',
        actions: [
          FrostedButton('Save', onPressed: () => setState(() => saved = true)),
        ],
      ),
      const SizedBox(height: 12),
      Text(saved ? 'Saved' : 'Unsaved'),
    ],
  );
}
