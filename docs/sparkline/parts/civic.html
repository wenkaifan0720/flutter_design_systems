Read a five-day trend

#

Compare two weeks of requests. The first ends at eight requests and the second ends at four. Read the Monday and Friday counts alongside the small line.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class ReadDailyTrend extends StatefulWidget {
  const ReadDailyTrend({super.key});
  @override
  State<ReadDailyTrend> createState() => _ReadDailyTrendState();
}

class _ReadDailyTrendState extends State<ReadDailyTrend> {
  bool alternate = false;

  List<double> get requests =>
      alternate ? const [9, 8, 6, 7, 4] : const [3, 5, 4, 6, 8];

  @override
  Widget build(BuildContext context) => SingleChildScrollView(
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 16,
      children: [
        CivicSegmentedControl<bool>(
          labelText: 'Show',
          segments: const [
            CivicSegment(value: false, label: 'First week'),
            CivicSegment(value: true, label: 'Second week'),
          ],
          value: alternate,
          onChanged: (next) => setState(() => alternate = next),
        ),
        CivicSparkline(label: 'Requests each day', values: requests),
        Text(
          'Monday ${requests.first.round()}, Friday ${requests.last.round()}',
        ),
      ],
    ),
  );
}
