import 'package:cupertino_ui/cupertino_ui.dart';

class DraftSections extends StatefulWidget {
  const DraftSections({super.key});

  @override
  State<DraftSections> createState() => _DraftSectionsState();
}

class _DraftSectionsState extends State<DraftSections> {
  int index = 0;

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 12,
    children: [
      CupertinoSlidingSegmentedControl<int>(
        groupValue: index,
        children: const {0: Text('Overview'), 1: Text('Activity')},
        onValueChanged: (v) {
          if (v != null) setState(() => index = v);
        },
      ),
      Text(index == 0 ? 'Draft summary' : 'Latest edits'),
    ],
  );
}

Parameters

CupertinoSlidingSegmentedControl API ↗
groupValue
