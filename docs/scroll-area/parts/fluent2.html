Scroll from Stop 1 to Stop 12 and back. The surrounding page stays in place while the stops move.

import 'package:flutter/widgets.dart';
import 'package:fluent2_design_ui/fluent2_design_ui.dart';

class BrowseStops extends StatefulWidget {
  const BrowseStops({super.key});
  @override
  State<BrowseStops> createState() => _BrowseStopsState();
}

class _BrowseStopsState extends State<BrowseStops> {
  final scroll = ScrollController();
  @override
  void dispose() {
    scroll.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 440),
    child: SizedBox(
      height: 150,
      child: Fluent2ScrollArea(
        controller: scroll,
        semanticLabel: 'Route stops',
        child: ListView(
          controller: scroll,
          children: [
            for (var stop = 1; stop <= 12; stop++)
              Padding(
                padding: const EdgeInsets.symmetric(vertical: 8),
                child: Text('Stop $stop'),
              ),
          ],
        ),
      ),
    ),
  );
}
