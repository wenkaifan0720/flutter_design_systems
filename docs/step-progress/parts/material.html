Review an order

#

Start at Cart, continue to Address and Review, then start again. The current step and its name change together.

components/step-progress/material/review_order.dart
import 'package:material_ui/material_ui.dart';

class ReviewOrder extends StatefulWidget {
  const ReviewOrder({super.key});

  @override
  State<ReviewOrder> createState() => _ReviewOrderState();
}

class _ReviewOrderState extends State<ReviewOrder> {
  int step = 0;

  @override
  Widget build(BuildContext context) => SizedBox(
    width: 480,
    height: 220,
    child: Stepper(
      currentStep: step,
      onStepContinue: () => setState(() => step = (step + 1) % 3),
      controlsBuilder: (context, details) => TextButton(
        onPressed: details.onStepContinue,
        child: Text(step == 2 ? 'Start again' : 'Continue'),
      ),
      steps: [
        for (final (index, name) in const ['Cart', 'Address', 'Review'].indexed)
          Step(title: Text(name), content: Text('Step ${index + 1}: $name')),
      ],
    ),
  );
}
