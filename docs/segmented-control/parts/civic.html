Choose a reporting period

#

Switch between Week and Month. The line below names the period included in the report.

Tab reaches the chosen answer; the arrow keys move to the next and choose it.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class ChoosePeriod extends StatefulWidget {
  const ChoosePeriod({super.key});
  @override
  State<ChoosePeriod> createState() => _ChoosePeriodState();
}

class _ChoosePeriodState extends State<ChoosePeriod> {
  String period = 'Week';

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 16,
    children: [
      CivicSegmentedControl<String>(
        labelText: 'Report period',
        segments: const [
          CivicSegment(value: 'Week', label: 'Week'),
          CivicSegment(value: 'Month', label: 'Month'),
        ],
        value: period,
        onChanged: (next) => setState(() => period = next),
      ),
      Text('Report period: $period'),
    ],
  );
}
