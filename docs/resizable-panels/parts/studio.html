import 'package:flutter/widgets.dart';
import 'package:studio_ui/studio_ui.dart';

class ResizePanes extends StatelessWidget {
  const ResizePanes({super.key});
  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 640),
    child: const SizedBox(
      height: 140,
      child: StudioSplitView(
        minFirst: 88,
        minSecond: 88,
        first: Center(child: Text('Files')),
        second: Center(child: Text('Notes')),
      ),
    ),
  );
}

Parameters

StudioSplitView API ↗
first
The start (or top) pane.
second
The end (or bottom) pane.
axis
Side by side, or stacked.
initialFraction
The first pane's share at the start.
minFirst
The minimum extent of the first pane.
minSecond
The minimum extent of the second pane.
onChanged
Called with the first pane's share as it moves.
semanticLabel
What a screen reader calls the grip.
