Find a note command

#

Open Commands and search for Review. Run Review note to change the note’s status. The Reset note command restores the draft.

Type to narrow the list; the up and down arrows move through it, Enter runs the command and Escape closes the palette.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';

class FindCommand extends StatefulWidget {
  const FindCommand({super.key});
  @override
  State<FindCommand> createState() => _FindCommandState();
}

class _FindCommandState extends State<FindCommand> {
  String status = 'Draft';

  @override
  Widget build(BuildContext context) => Column(
    mainAxisSize: MainAxisSize.min,
    spacing: 16,
    children: [
      CivicButton.secondary(
        expand: false,
        onPressed: () => showCivicCommandPalette(
          context: context,
          commands: [
            DesignCommand(
              label: 'Review note',
              group: 'Note',
              onPressed: () => setState(() => status = 'Reviewed'),
            ),
            DesignCommand(
              label: 'Reset note',
              group: 'Note',
              onPressed: () => setState(() => status = 'Draft'),
            ),
          ],
        ),
        child: const Text('Commands'),
      ),
      Text('Note: $status'),
    ],
  );
}
