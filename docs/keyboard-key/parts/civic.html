Reset a counter

#

Activate Count one a few times. With focus still in the example, press Escape to reset the count.

import 'package:civic_ui/civic_ui.dart';
import 'package:flutter/widgets.dart';
import 'package:flutter/services.dart';

class ResetCounter extends StatefulWidget {
  const ResetCounter({super.key});
  @override
  State<ResetCounter> createState() => _ResetCounterState();
}

class _ResetCounterState extends State<ResetCounter> {
  int count = 0;

  @override
  Widget build(BuildContext context) => CallbackShortcuts(
    bindings: {
      const SingleActivator(LogicalKeyboardKey.escape): () =>
          setState(() => count = 0),
    },
    child: Column(
      mainAxisSize: MainAxisSize.min,
      spacing: 16,
      children: [
        CivicButton.secondary(
          expand: false,
          onPressed: () => setState(() => count++),
          child: const Text('Count one'),
        ),
        Text.rich(
          TextSpan(
            children: [
              TextSpan(text: 'Count: $count. Press '),
              const WidgetSpan(
                alignment: PlaceholderAlignment.middle,
                child: CivicKbd('Esc', semanticLabel: 'Escape'),
              ),
              const TextSpan(text: ' to reset it.'),
            ],
          ),
        ),
      ],
    ),
  );
}

Parameters

label
The key’s name, as it is printed on the key.
