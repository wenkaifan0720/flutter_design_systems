import 'package:cupertino_ui/cupertino_ui.dart';

class ReadDeliveryNote extends StatelessWidget {
  const ReadDeliveryNote({super.key});

  @override
  Widget build(BuildContext context) => CupertinoButton(
    onPressed: () => showCupertinoSheet<void>(
      context: context,
      scrollableBuilder: (context, scrollController) => CupertinoPageScaffold(
        navigationBar: CupertinoNavigationBar(
          middle: const Text('Delivery note'),
          trailing: CupertinoButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Close'),
          ),
        ),
        child: SafeArea(
          child: ListView(
            controller: scrollController,
            padding: const EdgeInsets.all(20),
            children: const [Text('Leave the parcel at the studio desk.')],
          ),
        ),
      ),
    ),
    child: const Text('Delivery note'),
  );
}

Parameters
