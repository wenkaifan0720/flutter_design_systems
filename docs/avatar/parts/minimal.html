Switch reviewers

#

Switch between Ana Lee and Jo Ray. Compare the initials on the avatar with the full name beneath it.

components/avatar/minimal/switch_reviewer.dart
import 'package:flutter/widgets.dart';
import 'package:minimal_ui/minimal_ui.dart';

class SwitchReviewer extends StatefulWidget {
  const SwitchReviewer({super.key});
  @override
  State<SwitchReviewer> createState() => _SwitchReviewerState();
}

class _SwitchReviewerState extends State<SwitchReviewer> {
  bool ana = true;
  @override
  Widget build(BuildContext context) {
    final name = ana ? 'Ana Lee' : 'Jo Ray';
    return ConstrainedBox(
      constraints: const BoxConstraints(maxWidth: 440),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          MinimalAvatar(name),
          const SizedBox(height: 12),
          Text(name),
          const SizedBox(height: 12),
          MinimalButton(
            'Switch reviewer',
            onPressed: () => setState(() => ana = !ana),
          ),
        ],
      ),
    );
  }
}
