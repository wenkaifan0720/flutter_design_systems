import 'package:cupertino_ui/cupertino_ui.dart';

class ChooseDay extends StatefulWidget {
  const ChooseDay({super.key});
  @override
  State<ChooseDay> createState() => _ChooseDayState();
}

class _ChooseDayState extends State<ChooseDay> {
  int day = 12;

  @override
  Widget build(BuildContext context) => SizedBox(
    width: 280,
    child: Column(
      mainAxisSize: MainAxisSize.min,
      children: [
        SizedBox(
          height: 150,
          child: CupertinoPicker(
            itemExtent: 32,
            scrollController: FixedExtentScrollController(initialItem: day - 1),
            onSelectedItemChanged: (index) => setState(() => day = index + 1),
            children: [
              for (var date = 1; date <= 30; date++)
                Center(child: Text('November $date')),
            ],
          ),
        ),
        const SizedBox(height: 12),
        Text('November $day'),
      ],
    ),
  );
}

Parameters

CupertinoPicker API ↗
children
The rows of the wheel.
itemExtent
The height of each row.
onSelectedItemChanged
Called with the index of the row that settles in the middle.
scrollController
A FixedExtentScrollController that sets the first row shown.
looping
Whether the wheel wraps from the last row to the first.
useMagnifier
Whether the middle row is drawn larger.
