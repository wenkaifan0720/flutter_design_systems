Read draft details

#

Open the details and dismiss the dialog to return to its trigger. Changing the theme closes the dialog so you can reopen it in the new palette.

components/dialog/brutal/draft_details.dart
import 'package:flutter/widgets.dart';
import 'package:brutal_ui/brutal_ui.dart';

class DraftDetails extends StatelessWidget {
  const DraftDetails({super.key});

  @override
  Widget build(BuildContext context) => BrutalButton.filled(
    'View draft details',
    onPressed: () => showBrutalDialog<void>(
      context: context,
      builder: (dialogContext) => BrutalDialog(
        title: 'Draft details',
        content: const Text('Only visible to you until you publish'),
        actions: [
          BrutalDialogAction(
            label: 'Done',
            isDefaultAction: true,
            onPressed: () => Navigator.pop(dialogContext),
          ),
        ],
      ),
    ),
  );
}
