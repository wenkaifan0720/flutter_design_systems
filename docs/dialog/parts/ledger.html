Read draft details

#

Open the details and dismiss the dialog to return to its trigger. Changing the theme closes the dialog so you can reopen it in the new palette.

import 'package:flutter/widgets.dart';
import 'package:ledger_ui/ledger_ui.dart';

class DraftDetails extends StatelessWidget {
  const DraftDetails({super.key});

  @override
  Widget build(BuildContext context) => LedgerButton.filled(
    'View draft details',
    onPressed: () => LedgerDialog.show(
      context,
      title: 'Draft details',
      body: 'Only visible to you until you publish',
      type: LedgerDialogType.acknowledge,
      confirmLabel: 'Done',
    ),
  );
}
