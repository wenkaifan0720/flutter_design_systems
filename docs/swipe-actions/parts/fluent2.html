Archive a note

#

Reveal Archive from the trailing side of the Route note row. Archive it and use Restore note to bring it back.

import 'package:flutter/widgets.dart';
import 'package:fluent2_design_ui/fluent2_design_ui.dart';

class ArchiveNote extends StatefulWidget {
  const ArchiveNote({super.key});
  @override
  State<ArchiveNote> createState() => _ArchiveNoteState();
}

class _ArchiveNoteState extends State<ArchiveNote> {
  bool archived = false;
  bool read = false;
  @override
  Widget build(BuildContext context) => ConstrainedBox(
    constraints: const BoxConstraints(maxWidth: 440),
    child: archived
        ? Fluent2Button(
            'Restore note',
            onPressed: () => setState(() {
              archived = false;
              read = false;
            }),
          )
        : Fluent2SwipeActions(
            trailing: [
              Fluent2SwipeAction(
                label: 'Archive',
                icon: Fluent2Icons.archive,
                onPressed: () => setState(() => archived = true),
              ),
            ],
            child: Fluent2ListItem(
              title: 'Route note',
              subtitle: read ? 'Read' : 'Unread',
              onTap: () => setState(() => read = !read),
            ),
          ),
  );
}

Parameters

Fluent2SwipeActions API ↗
child
Supplies the content inside the component.
leading
The acts a drag toward the end shows, the first nearest the edge.
trailing
The acts a drag toward the start shows, the first nearest the edge.
fullSwipe
Whether a drag past 60% of the row does its edge's first act.
initiallyOpen
Starts the row with its actions revealed.
